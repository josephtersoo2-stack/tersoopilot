# Implementation Plan: Mobile Anti-Detect Browser (GeckoView Engine)

**System Target:** Native Android (Kotlin)

**Primary Engine:** Mozilla GeckoView (Standalone Engine)

**Target Trust Score:** $\ge 90\%$ on CreepJS, 100% "Reliable" on Iphey & Pixelscan

**Agent Directive:** Follow every architectural rule, directory hierarchy, and code implementation strictly. Do not substitute GeckoView with Android System WebView. Do not use generic random math jitter for fingerprint evasion.

---

## 1. Architectural Blueprint & Anti-Detect Principles

### Core Anti-Detection Rules

1. **Engine-Identity Alignment:** All profiles MUST report an authentic Mobile Firefox on Android User-Agent. Never spoof Blink/Chrome on Gecko. SpiderMonkey engine execution quirks, error stack formatting, and CSS capabilities will immediately trigger high-severity "Engine Discrepancy" flags on CreepJS if a Chrome UA is declared.
2. **Native C++ Protection via RFP:** Canvas, AudioContext, and high-precision system timers must be managed via Firefox's internal `privacy.resistFingerprinting` (RFP) engine rather than userland JavaScript overrides. RFP handles farbling in native C++ without detectable prototype patches.
3. **Strict Profile Sandboxing:** Every profile receives a dedicated filesystem root directory. Cookies, IndexedDB, local storage, HTTP cache, and session tokens must never cross profile boundaries.
4. **Pre-Execution Script Injection:** When dynamic overrides (such as hardware concurrency or screen properties) are applied, they must be injected via an internal WebExtension running at `run_at: "document_start"`. Never rely on late-running runtime injection.

```javascript
(() => {
  'use strict';

  browser.runtime.sendMessage({ type: "FETCH_CONFIG" }, (response) => {
    if (!response || !response.config) return;
    const cfg = response.config;

    // Inject override payload into the page context
    const code = `
      (() => {
        'use strict';
        const nativeToString = Function.prototype.toString;
        const registry = new WeakMap();

        function wrapNative(fn, name) {
          registry.set(fn, 'function ' + name + '() { [native code] }');
          return fn;
        }

        Function.prototype.toString = new Proxy(nativeToString, {
          apply(target, thisArg, args) {
            if (registry.has(thisArg)) {
              return registry.get(thisArg);
            }
            return Reflect.apply(target, thisArg, args);
          }
        });
        wrapNative(Function.prototype.toString, 'toString');

        // Navigator hardware concurrency override
        const origHC = Object.getOwnPropertyDescriptor(Navigator.prototype, 'hardwareConcurrency');
        if (origHC) {
          Object.defineProperty(Navigator.prototype, 'hardwareConcurrency', {
            get: wrapNative(() => ${cfg.hardwareConcurrency}, 'get hardwareConcurrency'),
            enumerable: true,
            configurable: true
          });
        }

        // WebGL vendor and renderer unmasking
        const hookWebGL = (proto) => {
          if (!proto) return;
          const origGetParam = proto.prototype.getParameter;
          proto.prototype.getParameter = wrapNative(function(param) {
            if (param === 0x9245) return '${cfg.webGlVendor}';
            if (param === 0x9246) return '${cfg.webGlRenderer}';
            return origGetParam.call(this, param);
          }, 'getParameter');
        };

        if (window.WebGLRenderingContext) hookWebGL(WebGLRenderingContext);
        if (window.WebGL2RenderingContext) hookWebGL(WebGL2RenderingContext);
      })();
    `;

    const scriptEl = document.createElement('script');
    scriptEl.textContent = code;
    (document.head || document.documentElement).appendChild(scriptEl);
    scriptEl.remove(); // Clean DOM to leave no traces
  });
})();
```

## 5. GeckoView Profile & Runtime Management

### File: `app/src/main/java/com/multibrowser/antidetect/engine/GeckoProfileManager.kt`

```kotlin
package com.multibrowser.antidetect.engine

import android.content.Context
import com.multibrowser.antidetect.models.BrowserProfile
import org.json.JSONObject
import org.mozilla.geckoview.GeckoRuntime
import org.mozilla.geckoview.GeckoRuntimeSettings
import org.mozilla.geckoview.GeckoSession
import org.mozilla.geckoview.GeckoSessionSettings
import org.mozilla.geckoview.WebExtension
import java.io.File
import java.io.FileWriter

class GeckoProfileManager(private val context: Context) {

    private var runtime: GeckoRuntime? = null
    private var nativePort: WebExtension.Port? = null

    fun initializeProfile(profile: BrowserProfile): Pair<GeckoRuntime, GeckoSession> {
        val profileDir = File(context.filesDir, "profiles/${profile.id}").apply {
            if (!exists()) mkdirs()
        }

        // 1. Generate geckoview-config.yaml for Native RFP & Security Settings
        val configFile = File(profileDir, "geckoview-config.yaml")
        writeProfileConfig(configFile)

        // 2. Build GeckoRuntime with isolated config path
        val runtimeSettings = GeckoRuntimeSettings.Builder()
            .configFilePath(configFile.absolutePath)
            .consoleOutput(false)
            .build()

        val activeRuntime = GeckoRuntime.create(context, runtimeSettings)
        this.runtime = activeRuntime

        // 3. Install Extension & Set Up Bridge
        registerExtensionBridge(activeRuntime, profile)

        // 4. Create Sandboxed GeckoSession with matching UA
        val sessionSettings = GeckoSessionSettings.Builder()
            .usePrivateMode(false)
            .userAgentOverride(profile.userAgent)
            .viewportMode(GeckoSessionSettings.VIEWPORT_MODE_MOBILE)
            .build()

        val session = GeckoSession(sessionSettings)
        session.open(activeRuntime)

        return Pair(activeRuntime, session)
    }

    private fun writeProfileConfig(targetFile: File) {
        val yaml = """
            env:
              MOZ_REMOTE_SETTINGS_DEV: "1"
            args:
              - "--pref"
              - "privacy.resistFingerprinting=true"
              - "--pref"
              - "privacy.resistFingerprinting.autoDeclineNoUserInputCanvasPrompts=true"
              - "--pref"
              - "privacy.reduceTimerPrecision=true"
              - "--pref"
              - "privacy.resistFingerprinting.reduceTimerPrecision.microseconds=20000"
              - "--pref"
              - "media.peerconnection.ice.default_address_only=true"
              - "--pref"
              - "media.peerconnection.ice.no_host=true"
              - "--pref"
              - "dom.security.https_only_mode=false"
        """.trimIndent()

        FileWriter(targetFile, false).use { it.write(yaml) }
    }

    private fun registerExtensionBridge(runtime: GeckoRuntime, profile: BrowserProfile) {
        val extensionUri = "resource://android/assets/extensions/antidetect/"

        runtime.webExtensionController.installBuiltIn(extensionUri)
            .accept(
                { extension ->
                    extension?.let { ext ->
                        runtime.webExtensionController.setMessageDelegate(
                            ext,
                            object : WebExtension.MessageDelegate {
                                override fun onConnect(port: WebExtension.Port) {
                                    nativePort = port
                                    pushProfileConfig(profile)
                                }
                            },
                            "antidetect_bridge"
                        )
                    }
                },
                { throwable ->
                    throwable.printStackTrace()
                }
            )
    }

    private fun pushProfileConfig(profile: BrowserProfile) {
        val payload = JSONObject().apply {
            put("action", "APPLY_PROFILE")
            put("payload", JSONObject().apply {
                put("hardwareConcurrency", profile.hardwareConcurrency)
                put("deviceMemory", profile.deviceMemory)
                put("webGlVendor", profile.webGlVendor)
                put("webGlRenderer", profile.webGlRenderer)
            })
        }
        nativePort?.postMessage(payload)
    }
}

```

### File: `app/src/main/java/com/multibrowser/antidetect/MainActivity.kt`

```kotlin
package com.multibrowser.antidetect

import android.os.Bundle
import android.view.inputmethod.EditorInfo
import android.widget.ArrayAdapter
import androidx.appcompat.app.AppCompatActivity
import com.multibrowser.antidetect.databinding.ActivityMainBinding
import com.multibrowser.antidetect.engine.GeckoProfileManager
import com.multibrowser.antidetect.models.BrowserProfile
import com.multibrowser.antidetect.models.ProfilePresets
import org.mozilla.geckoview.GeckoSession

class MainActivity : AppCompatActivity() {

    private lateinit var binding: ActivityMainBinding
    private lateinit var profileManager: GeckoProfileManager
    private var currentSession: GeckoSession? = null

    private val profileList = listOf(
        ProfilePresets.SAMSUNG_A54,
        ProfilePresets.PIXEL_8A,
        ProfilePresets.TECNO_CAMON_30
    )

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        binding = ActivityMainBinding.inflate(layoutInflater)
        setContentView(binding.root)

        profileManager = GeckoProfileManager(this)
        setupUI()
        loadProfile(profileList[0]) // Default: Samsung Galaxy A54
    }

    private fun setupUI() {
        val adapter = ArrayAdapter(
            this,
            android.R.layout.simple_spinner_dropdown_item,
            profileList.map { it.name }
        )
        binding.profileSpinner.adapter = adapter

        binding.btnSwitchProfile.setOnClickListener {
            val selectedProfile = profileList[binding.profileSpinner.selectedItemPosition]
            loadProfile(selectedProfile)
        }

        binding.btnGo.setOnClickListener {
            navigateFromInput()
        }

        binding.urlInput.setOnEditorActionListener { _, actionId, _ ->
            if (actionId == EditorInfo.IME_ACTION_GO) {
                navigateFromInput()
                true
            } else {
                false
            }
        }
    }

    private fun loadProfile(profile: BrowserProfile) {
        currentSession?.close()

        val (_, session) = profileManager.initializeProfile(profile)
        this.currentSession = session
        binding.geckoview.setSession(session)

        // Initial default diagnostic load
        session.loadUri("https://iphey.com")
        binding.urlInput.setText("https://iphey.com")
    }

    private fun navigateFromInput() {
        var input = binding.urlInput.text.toString().trim()
        if (input.isNotEmpty()) {
            if (!input.startsWith("http://") && !input.startsWith("https://")) {
                input = "https://$input"
            }
            currentSession?.loadUri(input)
        }
    }

    override fun onDestroy() {
        currentSession?.close()
        super.onDestroy()
    }
}
```
