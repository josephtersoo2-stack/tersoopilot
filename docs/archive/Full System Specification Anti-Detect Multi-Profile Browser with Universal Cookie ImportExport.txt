# Full System Specification: Anti-Detect Multi-Profile Browser with Universal Cookie Import/Export

This comprehensive specification covers the mobile GeckoView client, native SQLite ingestion/extraction routines, React Admin dashboard, and Django REST backend.

It incorporates full lifecycle management for cookies:

1. **Importing cookies onto pre-existing profiles** (re-populating live or dormant profiles).
2. **Pre-populating cookies during profile creation** (uploading or pasting Netscape/JSON cookies into the creation bottom sheet so the browser launches pre-authenticated).
3. **Exporting cookies from the Android app** (copying to clipboard or saving as standard Netscape/JSON `.json`/`.txt` files).
4. **Exporting cookies directly from the Django backend / React dashboard** (downloading any cloud-synced profile's session cookies straight to your desktop).

```python
        child=serializers.DictField(),
```

```python
import json
from django.http import HttpResponse
from rest_framework.views import APIView
from rest_framework.response import Response
from rest_framework import status, viewsets
from .models import SavedProfile, GlobalSetting, ProfileSyncEnvelope
from .serializers import (
    SavedProfileSerializer,
    CookieImportExportSerializer,
    GlobalSettingSerializer,
    ProfileSyncPushSerializer,
    ProfileSyncPullSerializer
)

class SavedProfileViewSet(viewsets.ModelViewSet):
    queryset = SavedProfile.objects.all().order_by("-updated_at")
    serializer_class = SavedProfileSerializer

class ProfileCookieExportView(APIView):
    """Exports cookies for a profile as a downloadable JSON file or raw JSON payload."""
    def get(self, request, profile_id):
        try:
            profile = SavedProfile.objects.get(id=profile_id)
            cookies_data = json.loads(profile.stored_cookies_json or "[]")

            # Check if user requested direct file attachment
            as_file = request.query_params.get("download", "false").lower() == "true"
            if as_file:
                response = HttpResponse(
                    json.dumps(cookies_data, indent=2),
                    content_type="application/json"
                )
                response["Content-Disposition"] = f'attachment; filename="cookies_{profile.name.replace(" ", "_")}.json"'
                return response

            return Response({
                "profile_id": str(profile.id),
                "cookie_count": len(cookies_data),
                "cookies": cookies_data
            }, status=status.HTTP_200_OK)
        except SavedProfile.DoesNotExist:
            return Response({"error": "Profile not found"}, status=status.HTTP_404_NOT_FOUND)

class ProfileCookieImportView(APIView):
    """Imports raw Netscape/JSON cookies directly into a backend profile record."""
    def post(self, request, profile_id):
        try:
            profile = SavedProfile.objects.get(id=profile_id)
            serializer = CookieImportExportSerializer(data=request.data)
            if not serializer.is_valid():
                return Response(serializer.errors, status=status.HTTP_400_BAD_REQUEST)

            cookie_list = serializer.validated_data["cookies"]
            profile.stored_cookies_json = json.dumps(cookie_list)
            profile.cookie_count = len(cookie_list)
            profile.save()

            return Response({
                "status": "SUCCESS",
                "profile_id": str(profile.id),
                "imported_count": profile.cookie_count
            }, status=status.HTTP_200_OK)
        except SavedProfile.DoesNotExist:
            return Response({"error": "Profile not found"}, status=status.HTTP_404_NOT_FOUND)

class GlobalSettingView(APIView):
    def get(self, request):
        return Response(GlobalSettingSerializer(GlobalSetting.load()).data)

    def patch(self, request):
        serializer = GlobalSettingSerializer(GlobalSetting.load(), data=request.data, partial=True)
        if serializer.is_valid():
            serializer.save()
            return Response(serializer.data)
        return Response(serializer.errors, status=status.HTTP_400_BAD_REQUEST)

class ProfileSyncView(APIView):
    def post(self, request, profile_id):
        serializer = ProfileSyncPushSerializer(data=request.data)
        if not serializer.is_valid():
            return Response(serializer.errors, status=status.HTTP_400_BAD_REQUEST)

        owner = request.user.username if request.user.is_authenticated else "default_user"
        v = serializer.validated_data

        env, created = ProfileSyncEnvelope.objects.get_or_create(
            profile_id=profile_id,
            owner_username=owner,
            defaults={**v, "version": 1}
        )
        if not created:
            for key, val in v.items():
                setattr(env, key, val)
            env.version += 1
            env.save()

        return Response({"status": "SYNCED", "version": env.version}, status=status.HTTP_200_OK)

    def get(self, request, profile_id):
        owner = request.user.username if request.user.is_authenticated else "default_user"
        try:
            env = ProfileSyncEnvelope.objects.get(profile_id=profile_id, owner_username=owner)
            return Response(ProfileSyncPullSerializer(env).data)
        except ProfileSyncEnvelope.DoesNotExist:
            return Response({"error": "Backup not found"}, status=status.HTTP_404_NOT_FOUND)
```

## 5. Android Client: Universal SQLite Cookie Ingestion & Extraction Engine

This engine operates with **zero database locking issues**. It safely parses Netscape/JSON cookies, writes directly to `moz_cookies` with correct Firefox schema constraints, and handles conversions between Netscape and modern JSON.

### File: `app/src/main/java/com/multibrowser/antidetect/sync/CookieEngine.kt`

```kotlin
package com.multibrowser.antidetect.sync

import android.content.ContentValues
import android.content.Context
import android.database.sqlite.SQLiteDatabase
import org.json.JSONArray
import org.json.JSONObject
import java.io.File

object CookieEngine {

    /**
     * Extracts all cookies from a profile's isolated cookies.sqlite database into clean JSON.
     */
    fun exportCookiesToJson(context: Context, profileId: String): String {
        val profileDir = File(context.filesDir, "profiles/$profileId")
        val dbFile = File(profileDir, "cookies.sqlite")

        if (!dbFile.exists()) return "[]"

        val jsonArray = JSONArray()
        val db = SQLiteDatabase.openDatabase(dbFile.absolutePath, null, SQLiteDatabase.OPEN_READONLY)
        try {
            val cursor = db.rawQuery(
                """
                SELECT host, name, value, path, expiry, isSecure, isHttpOnly, sameSite 
                FROM moz_cookies
                """.trimIndent(),
                null
            )
            while (cursor.moveToNext()) {
                val cookie = JSONObject().apply {
                    put("domain", cursor.getString(0))
                    put("host", cursor.getString(0))
                    put("name", cursor.getString(1))
                    put("value", cursor.getString(2))
                    put("path", cursor.getString(3))
                    put("expiry", cursor.getLong(4))
                    put("isSecure", cursor.getInt(5) == 1)
                    put("isHttpOnly", cursor.getInt(6) == 1)
                    put("sameSite", cursor.getInt(7))
                }
                jsonArray.put(cookie)
            }
            cursor.close()
        } finally {
            db.close()
        }
        return jsonArray.toString(2)
    }

    /**
     * Imports a JSON or Netscape-style cookie array into an existing or uninitialized profile directory.
     */
    fun importCookiesFromJson(context: Context, profileId: String, rawJsonOrNetscape: String): Int {
        val profileDir = File(context.filesDir, "profiles/$profileId").apply {
            if (!exists()) mkdirs()
        }

        // 1. Flush any orphan WAL files that would cause SQLite locks
        File(profileDir, "cookies.sqlite-wal").delete()
        File(profileDir, "cookies.sqlite-shm").delete()

        // 2. Normalize input: parse either standard JSON array or line-delimited Netscape format
        val cookiesArray = parseToStandardJsonArray(rawJsonOrNetscape)

        val dbFile = File(profileDir, "cookies.sqlite")
        val db = SQLiteDatabase.openOrCreateDatabase(dbFile, null)
        var importedCount = 0

        try {
            // Guarantee GeckoView moz_cookies table exists
            db.execSQL(
                """
                CREATE TABLE IF NOT EXISTS moz_cookies (
                    id INTEGER PRIMARY KEY,
                    originAttributes TEXT NOT NULL DEFAULT '',
                    name TEXT,
                    value TEXT,
                    host TEXT,
                    path TEXT,
                    expiry INTEGER,
                    lastAccessed INTEGER,
                    creationTime INTEGER,
                    isSecure INTEGER,
                    isHttpOnly INTEGER,
                    inBrowserElement INTEGER DEFAULT 0,
                    sameSite INTEGER DEFAULT 0,
                    rawSameSite INTEGER DEFAULT 0,
                    schemeMap INTEGER DEFAULT 0
                );
                """.trimIndent()
            )
            db.execSQL("CREATE UNIQUE INDEX IF NOT EXISTS moz_basedomain ON moz_cookies (host, name, path, originAttributes);")

            db.beginTransaction()
            val nowMicroseconds = System.currentTimeMillis() * 1000

            for (i in 0 until cookiesArray.length()) {
                val c = cookiesArray.getJSONObject(i)
                val domain = c.optString("domain", c.optString("host", ""))
                val name = c.optString("name", "")
                val value = c.optString("value", "")
                val path = c.optString("path", "/")

                if (domain.isBlank() || name.isBlank()) continue

                // Expiry calculation: default to 1 year in seconds if missing
                val expiry = c.optLong("expiry", (System.currentTimeMillis() / 1000) + 31536000)

                val cv = ContentValues().apply {
                    put("host", domain)
                    put("name", name)
                    put("value", value)
                    put("path", path)
                    put("expiry", expiry)
                    put("isSecure", if (c.optBoolean("isSecure", false)) 1 else 0)
                    put("isHttpOnly", if (c.optBoolean("isHttpOnly", false)) 1 else 0)
                    put("sameSite", c.optInt("sameSite", 0))
                    put("lastAccessed", nowMicroseconds)
                    put("creationTime", nowMicroseconds)
                }

                db.insertWithOnConflict("moz_cookies", null, cv, SQLiteDatabase.CONFLICT_REPLACE)
                importedCount++
            }
            db.setTransactionSuccessful()
        } finally {
            db.endTransaction()
            db.close()
        }

        return importedCount
    }

    private fun parseToStandardJsonArray(raw: String): JSONArray {
        val trimmed = raw.trim()
        if (trimmed.startsWith("[") && trimmed.endsWith("]")) {
            return JSONArray(trimmed)
        }

        // Netscape format fallback: domain \t flag \t path \t secure \t expiry \t name \t value
        val result = JSONArray()
        trimmed.lines().forEach { line ->
            val l = line.trim()
            if (l.isNotEmpty() && !l.startsWith("#")) {
                val parts = l.split("\t")
                if (parts.size >= 7) {
                    val obj = JSONObject().apply {
                        put("domain", parts[0])
                        put("path", parts[2])
                        put("isSecure", parts[3].equals("TRUE", ignoreCase = true))
                        put("expiry", parts[4].toLongOrNull() ?: ((System.currentTimeMillis() / 1000) + 31536000))
                        put("name", parts[5])
                        put("value", parts[6])
                        put("isHttpOnly", false)
                        put("sameSite", 0)
                    }
                    result.put(obj)
                }
            }
        }
        return result
    }
}
```

```kotlin
            Column(modifier = Modifier.fillMaxWidth(), verticalArrangement = Arrangement.spacedBy(12.dp)) {
                TabRow(selectedTabIndex = tabIndex) {
                    Tab(selected = tabIndex == 0, onClick = { tabIndex = 0 }, text = { Text("Export") })
                    Tab(selected = tabIndex == 1, onClick = { tabIndex = 1 }, text = { Text("Import") })
                }

                if (tabIndex == 0) {
                    val exported = remember { CookieEngine.exportCookiesToJson(context, profileId) }
                    OutlinedTextField(
                        value = exported,
                        onValueChange = {},
                        readOnly = true,
                        modifier = Modifier.fillMaxWidth().height(180.dp),
                        textStyle = MaterialTheme.typography.bodySmall
                    )
                    Button(
                        onClick = {
                            val clipboard = context.getSystemService(Context.CLIPBOARD_SERVICE) as ClipboardManager
                            clipboard.setPrimaryClip(ClipData.newPlainText("Cookies", exported))
                            Toast.makeText(context, "Copied to Clipboard!", Toast.LENGTH_SHORT).show()
                            onDismiss()
                        },
                        modifier = Modifier.fillMaxWidth()
                    ) {
                        Text("Copy to Clipboard")
                    }
                } else {
                    OutlinedTextField(
                        value = pastedJson,
                        onValueChange = { pastedJson = it },
                        placeholder = { Text("Paste JSON or Netscape cookies here...") },
                        modifier = Modifier.fillMaxWidth().height(180.dp),
                        textStyle = MaterialTheme.typography.bodySmall
                    )
                    Button(
                        onClick = {
                            if (pastedJson.isNotBlank()) {
                                val count = CookieEngine.importCookiesFromJson(context, profileId, pastedJson)
                                Toast.makeText(context, "Imported $count cookies!", Toast.LENGTH_SHORT).show()
                                onCookiesUpdated(count)
                                onDismiss()
                            }
                        },
                        modifier = Modifier.fillMaxWidth()
                    ) {
                        Text("Import & Overwrite")
                    }
                }
            }
        },
        confirmButton = {},
        dismissButton = {
            TextButton(onClick = onDismiss) { Text("Close") }
        }
    )
}
```
